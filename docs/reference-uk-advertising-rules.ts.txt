/**
 * COMPLIANCE LEXICON
 *
 * Every entry below is derived from a published ASA ruling or CAP AdviceOnline
 * guidance. Cite the source when adding an entry. Do not soften or remove an
 * entry without a documented regulatory reason recorded in docs/02-compliance.md.
 *
 * Source: CAP AdviceOnline "Weight control: Prescription-only medicines"
 * (updated 29 May 2026) and "Healthcare: Prescription-only Medicines (websites)".
 */

export type Severity = "block" | "warn";

export interface LexiconRule {
  /** Regex, case-insensitive, word-boundary aware. */
  pattern: RegExp;
  /** Human label for the scanner output. */
  label: string;
  /** Zones in which this rule is enforced. */
  zones: Array<"A" | "B" | "C">;
  severity: Severity;
  /** Why it is prohibited — printed in the failure message. */
  reason: string;
  /** ASA ruling or CAP advice reference. */
  source: string;
  /** What to write instead. */
  fix: string;
}
